package com.techchallenger.oficina360.mappers;

import com.techchallenger.oficina360.dominio.OrdemServico;
import com.techchallenger.oficina360.dtos.ordemservico.CriarOrdemServicoRequestDTO;
import com.techchallenger.oficina360.frameworks.persistence.entities.OrdemServicoEntity;

import static com.techchallenger.oficina360.utils.FormataDadosUtils.mascararPlaca;

public class CriarOrdemServicoMapper {

    private CriarOrdemServicoMapper() {
    }

    public static CriarOrdemServicoRequestDTO toDTO(OrdemServico ordemServico) {
        if (ordemServico == null) {
            return null;
        }

        return new CriarOrdemServicoRequestDTO(
//                ordemServico.getId(),
                ordemServico.getDocumentoCliente(),
                mascararPlaca(ordemServico.getPlacaVeiculo()),
                ordemServico.getDescricaoProblema()
//                ordemServico.getOrdemDeServicoStatus()
        );
    }

    public static OrdemServicoEntity toEntity(CriarOrdemServicoRequestDTO criarOrdemServicoRequestDTO) {
        if (criarOrdemServicoRequestDTO == null) {
            return null;
        }

        return OrdemServicoEntity.builder()
//                .id(criarOrdemServicoRequestDTO.id())
                .documentoCliente(criarOrdemServicoRequestDTO.documentoCliente())
                .placaVeiculo(criarOrdemServicoRequestDTO.placaVeiculo())
                .descricaoProblema(criarOrdemServicoRequestDTO.descricaoProblema())
//                .ordemDeServicoStatus(criarOrdemServicoRequestDTO.ordemDeServicoStatus())
                .build();
    }

    public static void updateEntityFromDto(
            CriarOrdemServicoRequestDTO criarOrdemServicoRequestDTO,
            OrdemServicoEntity ordemServicoEntity
    ) {
        if (criarOrdemServicoRequestDTO == null || ordemServicoEntity == null) {
            return;
        }

        ordemServicoEntity.setDocumentoCliente(criarOrdemServicoRequestDTO.documentoCliente());
        ordemServicoEntity.setPlacaVeiculo(criarOrdemServicoRequestDTO.placaVeiculo());
        ordemServicoEntity.setDescricaoProblema(criarOrdemServicoRequestDTO.descricaoProblema());
    }
}
